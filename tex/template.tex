\begin{exercise}{3.0}
  Restate the exercise here, e.g.\ show that every $\R$-algebra
  homomorphism $\F \to \R$ is determined by its kernel.
\end{exercise}

\begin{solution}
  Let $a \in \dual{\F}$ \dots
  \[
    \begin{tikzcd}
      \F \arrow[r, "a"] \arrow[d] & \R \\
      \F/\ker a \arrow[ur, "\sim"']
    \end{tikzcd}
  \]
\end{solution}
