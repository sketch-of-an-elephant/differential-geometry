%  Annotated book -- Nestruev, "Smooth Manifolds and Observables"
%  Differential Geometry Reading Group
%
%  Builds a copy of the book with the group's notes typeset in a margin
%  next to the page they refer to. You don't need to edit this file:
%  notes go in annotations/chNN.tex, e.g.
%
%      \bookNote{37}{Here $\dual{\F}$ is the set of all $\R$-points.}
%      \bookNote[Aleksandras]{38}{A longer remark, with display math
%        \[ \Spec_\R \F = \dual{\F}. \] }
%
%  The first argument is the page number PRINTED in the book (the arabic
%  numbers, starting with page 1 of Chapter 1). Pages with notes get a
%  wide margin on the right; all other pages are copied unchanged.
%  Notes that don't fit in the margin continue on an extra page inserted
%  right after the book page.
% =====================================================================

% \bookNote[author]{printed page}{text}: store the note for that page.
\newcommand{\bookNote}[3][]{%
  \@ifundefined{dg@notes@#2}{\@namedef{dg@notes@#2}{}}{}%
  \expandafter\g@addto@macro\csname dg@notes@#2\endcsname{\dg@onenote{#1}{#3}}%
  \ignorespaces}
\newcommand{\dg@onenote}[2]{%
  \par\noindent{\color{noterule}\rule[0.5ex]{\hsize}{0.4pt}}\par
  \if\relax\detokenize{#1}\relax\else
    \noindent{\footnotesize\bfseries\color{noterule!70!black}#1}\par\nobreak
  \fi
  #2\par}

% Collect every chapter's notes.
\newcount\dg@ch
\dg@ch=0
\loop\ifnum\dg@ch<30
  \advance\dg@ch by 1
  \InputIfFileExists{ch\ifnum\dg@ch<10 0\fi\the\dg@ch.tex}{}{}%
\repeat

% Ship out raw pages: origin at the top-left corner of the paper.
\hoffset=-1in \voffset=-1in
\pdfximage{\bookfile}
\newcount\dg@npages \dg@npages=\pdflastximagepages
\newcount\dg@p
\newcount\dg@printed
\newbox\dg@book
\newbox\dg@notes
\newbox\dg@chunk
\newdimen\dg@colht

% Typeset the notes for the current page into \dg@notes.
\newcommand{\dg@setnotes}{%
  \setbox\dg@notes\vbox{%
    \hsize=\notewidth \linewidth=\hsize \columnwidth=\hsize
    \parindent=0pt \parskip=3pt plus 1pt
    \small \raggedright
    \abovedisplayskip=4pt \belowdisplayskip=4pt
    \@nameuse{dg@notes@\the\dg@printed}}}

% One widened page: #1 = left content (book page or empty), the rest of
% the notes box is split to the column height.
\newcommand{\dg@shipwide}[2]{%
  \dg@colht=\ht\dg@book \advance\dg@colht by -2\notegap
  \advance\dg@colht by -16pt
  \vbadness=10000 \splittopskip=0pt
  \setbox\dg@chunk\vsplit\dg@notes to \dg@colht
  \setbox\dg@chunk\vbox{\unvbox\dg@chunk}% natural height, no stretching
  \pdfpagewidth=\wd\dg@book \advance\pdfpagewidth by \notewidth
  \advance\pdfpagewidth by 2\notegap
  \pdfpageheight=\ht\dg@book
  \shipout\hbox{%
    \ifx#1\relax\hbox to \wd\dg@book{\hfil}\else\copy\dg@book\fi
    \vbox to \ht\dg@book{%
      \hsize=\dimexpr\notewidth+2\notegap\relax
      \hrule height 0pt
      \hbox to \hsize{%
        \color{notebg}\vrule width \hsize height 0pt depth \ht\dg@book\hss}%
      \vskip-\ht\dg@book
      \vskip\notegap
      \moveright\notegap\hbox{\normalcolor\footnotesize\bfseries
        Notes on p.~\the\dg@printed#2}%
      \vskip 4pt
      \moveright\notegap\box\dg@chunk
      \vfil}}}

\newcommand{\dg@page}{%
  \dg@printed=\dg@p \advance\dg@printed by -\bookoffset
  \setbox\dg@book\hbox{\includegraphics[page=\the\dg@p]{\bookfile}}%
  \@ifundefined{dg@notes@\the\dg@printed}{%
    \pdfpagewidth=\wd\dg@book \pdfpageheight=\ht\dg@book
    \shipout\copy\dg@book}{%
    \dg@setnotes
    \dg@shipwide\dg@book{}%
    {\loop\ifvoid\dg@notes\else
       \dg@shipwide\relax{ (continued)}%
     \repeat}}}

\newcommand{\shipbook}{%
  \dg@p=0
  \loop\ifnum\dg@p<\dg@npages
    \advance\dg@p by 1
    \dg@page
  \repeat}
\makeatother
